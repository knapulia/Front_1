class Direction {
  static _nextId = 1;

  constructor(name, department, governmentConnection, usersDay1, usersDay2, implementationTerm, termUnit) {
    this.id = Direction._nextId++;
    this.name = name;
    this.department = department;
    this.governmentConnection = governmentConnection;
    this.usersDay1 = usersDay1;                  
    this.usersDay2 = usersDay2;                  
    this.implementationTerm = implementationTerm; 
    this.termUnit = termUnit;                   
  }

  _getTermInDays() {
    const termInDays = {
      'днів': 1,
      'місяців': 30,
      'років': 365,
      'тижнів': 7
    };
    return this.implementationTerm * (termInDays[this.termUnit] || 1);
  }

  getInfo() {
    return `ID: ${this.id}, Назва: ${this.name}, Відділ: ${this.department}, Термін: ${this.implementationTerm} ${this.termUnit}`;
  }
}

function sortDirectionsByTermAndCalcAverage() {
  const sortedDirections = [...directions];

  sortedDirections.sort((a, b) => a._getTermInDays() - b._getTermInDays());
  const groupByTerm = {};

  sortedDirections.forEach(dir => {
    // використовуємо ключ у форматі "число одиниця"
    const termKey = `${dir.implementationTerm}${dir.termUnit}`;

    if (!groupByTerm[termKey]) {
      // якщо групи ще не існує, створюємо її
      groupByTerm[termKey] = {
        directions: [],
        totalUsers: 0
      };
    }

    // додаємо напрям до групи та сумуємо користувачів
    groupByTerm[termKey].directions.push(dir);
    groupByTerm[termKey].totalUsers += dir.usersDay1 + dir.usersDay2;
  });

  // розраховуємо середню для кожної групи
  console.log('=== ГРУПУВАННЯ ЗА ТЕРМІНОМ З СЕРЕДНЬОЮ КІЛЬКІСТЮ КОРИСТУВАЧІВ ===');
  for (let termKey in groupByTerm) {
    const group = groupByTerm[termKey];
    // ділимо загальне число користувачів на
    const avgUsers = group.totalUsers / (group.directions.length * 2);
    console.log(`${termKey}: середня = ${avgUsers.toFixed(2)} користувачів`);
  }
}

function calculateCombinedDuration(directionIds) {
  console.log('\n=== ОБЧИСЛЕННЯ ТРИВАЛОСТІ ДЕКІЛЬКОХ НАПРЯМІВ ===');

  // знаходимо напрями з переданими id
  const selectedDirections = directions.filter(dir => directionIds.includes(dir.id));

  // групуємо за відділами
  const departmentGroups = {};

  selectedDirections.forEach(dir => {
    if (!departmentGroups[dir.department]) {
      departmentGroups[dir.department] = [];
    }
    departmentGroups[dir.department].push(dir);
  });

  // Для кожного відділу обчислюємо тривалість
  for (let deptName in departmentGroups) {
    const deptDirections = departmentGroups[deptName];
    let totalDuration = 0;

    // Цикл forEach для обчислення тривалості
    deptDirections.forEach((dir, index) => {
      const baseDuration = dir._getTermInDays();

      if (index === 0) {
        // Перший напрям, його базова тривалість
        totalDuration += baseDuration;
      } else {
        // кожен наступний напрям збільшує тривалість на 1.1 від його терміну
        totalDuration += baseDuration * 1.1;
      }
    });

    console.log(`Відділ: ${deptName}`);
    console.log(`Загальна тривалість: ${totalDuration.toFixed(1)} днів`);
  }
}

КОМЕНТАРІ:
- filter() знаходить напрями за умовою
- forEach() використовується для ітерації з індексом
- Логіка множення на 1.1 для додаткових напрямів
- Групування по дротам знаходиться через об'єкт departmentGroups

// Клас для представлення користувача
class User {
  constructor(surname, name, age, education, feedbackReason, contactDate, contactTime) {
    this.surname = surname;
    this.name = name;
    this.age = age;
    this.education = education;        // Тип освіти: 'вища', 'середня', 'без освіти'
    this.feedbackReason = feedbackReason;
    this.contactDate = new Date(contactDate); // Об'єкт Date для роботи з датами
    this.contactTime = contactTime;    // Час у форматі "HH:MM"
  }

  // приватний метод для отримання години контакту як числа
  _getContactHour() {
    const parts = this.contactTime.split(':');
    return parseInt(parts[0]);
  }

  // метод для отримання повного імені
  getFullName() {
    return `${this.surname} ${this.name}`;
  }

  // метод для виведення інформації
  getInfo() {
    return `${this.getFullName()}, вік: ${this.age}, освіта: ${this.education}`;
  }
}

function filterUsersByMonthAndTime(month, hour) {
  console.log('\n=== КОРИСТУВАЧІ ЗА МІСЯЦЕМ І ЧАСОМ ===');
  console.log(`Шукаємо користувачів місяця ${month} о ${hour}:00`);

  //використовуємо filter для знаходження користувачів
  const filtered = users.filter(user => {
    // getMonth() повертає 0-11, тому додаємо 1
    const userMonth = user.contactDate.getMonth() + 1;
    const userHour = user._getContactHour();
    // Повертаємо true, якщо обидві умови виконані
    return userMonth === month && userHour === hour;
  });

  if (filtered.length === 0) {
    console.log('Користувачів не знайдено');
  } else {
    filtered.forEach(user => console.log(`  - ${user.getInfo()}`));
  }

  return filtered;
}

КОМЕНТАРІ:
- filter() повертає новий масив з елементами, які задовольняють умову

- Умова використовує логічний оператор && для комбінування обох перевірок

function classifyUsers() {
  console.log('\n=== КЛАСИФІКАЦІЯ КОРИСТУВАЧІВ ===');

  const classes = {
    highEducationWorkHours: [],     // Клас 1: Вища освіта в робочий час
    noEducationOffHours: [],        // Клас 2: Без освіти в неробочий час
    others: []                       // Клас 3: Інші
  };

  // Цикл do-while для перевірки кожного користувача
  let index = 0;
  do {
    const user = users[index];
    const userHour = user._getContactHour();
    // Робочий час 09:00 до 17:00
    const isWorkHours = userHour >= 9 && userHour <= 17;

    // Розподілимо користувача на один з трьох класів
    if (user.education === 'вища' && isWorkHours) {
      // Якщо вища освіта АБО в робочий час
      classes.highEducationWorkHours.push(user);
    } else if (user.education === 'без освіти' && !isWorkHours) {
      // Якщо без освіти АБО в неробочий час (!isWorkHours)
      classes.noEducationOffHours.push(user);
    } else {
      // Всі інші користувачі
      classes.others.push(user);
    }

    index++;
  } while (index < users.length);

  //результати
  console.log(`1. Вища освіта в робочий час: ${classes.highEducationWorkHours.length}`);
  console.log(`2. Без освіти в неробочий час: ${classes.noEducationOffHours.length}`);
  console.log(`3. Інші: ${classes.others.length}`);

  return classes;
}

КОМЕНТАРІ:
- Використано цикл do-while для першої перевірки до умови
- Логічний оператор && комбінує дві умови (освіта І час)
- Оператор ! інвертує значення (не робочий час)
- Умовна конструкція if-else if-else розподіляє користувачів на класи

function sortUsersByNameAndShowFeedback() {
  console.log('\n=== КОРИСТУВАЧІ В АЛФАВІТНОМУ ПОРЯДКУ ===');

  // Копіюємо масив і впорядковуємо за прізвищем, потім за іменем
  const sortedUsers = [...users].sort((a, b) => {
    // порівнюємо прізвища
    const surnameCompare = a.surname.localeCompare(b.surname);

    if (surnameCompare !== 0) {
      // Якщо прізвища різні, повертаємо результат порівняння
      return surnameCompare;
    }

    // Якщо прізвища однакові, порівнюємо імена
    return a.name.localeCompare(b.name);
  });

  // Виводимо користувачів з метою зворотного зв'язку
  sortedUsers.forEach(user => {
    console.log(`${user.getFullName()} - Мета: ${user.feedbackReason}`);
  });

  return sortedUsers;
}

КОМЕНТАРІ:
- Спред оператор [...users] створює копію масиву
- Метод sort() з компаратором забезпечує спеціальне сортування
